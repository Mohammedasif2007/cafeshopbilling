# Asif Cafe Billing System (HTML, CSS, JavaScript)

A complete responsive billing website for a tea shop with two login roles:

* **Employee Login** → Create customer bills and print them.
* **Owner Login** → View today's sales summary and all generated bills.

This project uses only **HTML, CSS, and JavaScript** and stores data in the browser using `localStorage`.

# 6️⃣ js/app.js

```javascript
function login() {
  const role = document.getElementById('role').value;
  const username = document.getElementById('username').value;
  const password = document.getElementById('password').value;
  const error = document.getElementById('error');

  if (role === 'employee' && username === 'employee' && password === '1234') {
    localStorage.setItem('role', 'employee');
    window.location.href = 'employee.html';
  } else if (role === 'owner' && username === 'owner' && password === '1234') {
    localStorage.setItem('role', 'owner');
    window.location.href = 'owner.html';
  } else {
    error.textContent = 'Invalid username or password';
  }
}
```

```javascript
  { name: 'Veg Puff', price: 20, image: 'assets/products/veg-puff.png' },
  { name: 'Egg Puff', price: 25, image: 'assets/products/egg-puff.png' },
  { name: 'Vada', price: 10, image: 'assets/products/vada.png' },
  { name: 'Bajji', price: 15, image: 'assets/products/bajji.png' },
  { name: 'Biscuit', price: 10, image: 'assets/products/biscuit.png' },
  { name: 'Cake Slice', price: 30, image: 'assets/products/cake.png' },
  { name: 'Water Bottle', price: 20, image: 'assets/products/water.png' }
];

let cart = [];

function logout() {
  localStorage.removeItem('role');
  window.location.href = 'index.html';
}

function renderProducts(filteredProducts = products) {
  const productList = document.getElementById('productList');
  productList.innerHTML = '';

  filteredProducts.forEach((product) => {
    const index = products.findIndex(p => p.name === product.name);
    const div = document.createElement('div');
    div.className = 'product-card';
    div.innerHTML = `
      <img src="${product.image}" alt="${product.name}" class="product-image" onerror="this.src='assets/products/placeholder.png'">
      <h3>☕ ${product.name}</h3>
      <p>₹${product.price}</p>
      <button onclick="addToCart(${index})">Add</button>
    `;
    productList.appendChild(div);
  });
}

function addToCart(index) {
  const product = products[index];
  const existing = cart.find(item => item.name === product.name);

  if (existing) {
    existing.qty++;
  } else {
    cart.push({ ...product, qty: 1 });
  }

  renderCart();
}

function renderCart() {
  const cartItems = document.getElementById('cartItems');
  const totalEl = document.getElementById('total');

  cartItems.innerHTML = '';
  let total = 0;

  cart.forEach(item => {
    total += item.price * item.qty;

    const div = document.createElement('div');
    div.className = 'cart-item';
    div.innerHTML = `
      <span>${item.name} x ${item.qty}</span>
      <span>₹${item.price * item.qty}</span>
    `;
    cartItems.appendChild(div);
  });

  totalEl.textContent = total;
}

function finishOrder() {
  if (cart.length === 0) {
    alert('Please add at least one item.');
    return;
  }

  const total = cart.reduce((sum, item) => sum + item.price * item.qty, 0);

  const bill = {
    items: cart,
    total,
    date: new Date().toLocaleString()
  };

  localStorage.setItem('currentBill', JSON.stringify(bill));

  const sales = JSON.parse(localStorage.getItem('sales')) || [];
  sales.push(bill);
  localStorage.setItem('sales', JSON.stringify(sales));

  window.location.href = 'bill.html';
}

function searchProducts() {
  const keyword = document.getElementById('searchInput').value.toLowerCase();
  const filtered = products.filter(product =>
    product.name.toLowerCase().includes(keyword)
  );
  renderProducts(filtered);
}

renderProducts();
```

# js/monthly-report.js

```javascript
function toggleDarkMode() {
  document.body.classList.toggle('dark-mode');
  localStorage.setItem('darkMode', document.body.classList.contains('dark-mode'));
}

if (localStorage.getItem('darkMode') === 'true') {
  document.body.classList.add('dark-mode');
}

const sales = JSON.parse(localStorage.getItem('sales')) || [];
const now = new Date();

const monthlySales = sales.filter(sale => {
  const d = new Date(sale.date);
  return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear();
});

const total = monthlySales.reduce((sum, sale) => sum + sale.total, 0);

document.getElementById('monthTotal').textContent = `₹${total}`;
document.getElementById('monthBills').textContent = monthlySales.length;

const monthList = document.getElementById('monthList');
monthList.innerHTML = monthlySales.length
  ? monthlySales.map((sale, i) => `
      <div class="transaction">
        <strong>Bill #${i + 1}</strong><br>
        ₹${sale.total}<br>
        ${sale.date}
      </div>`).join('')
  : '<p>No sales this month.</p>';
```
